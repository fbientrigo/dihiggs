# 1. (Ya hecho) Construir la imagen en la base de datos de Docker
docker build -f .devcontainer/Dockerfile -t dihiggs-base .

# 2. [EL PASO FALTANTE] Exportar la imagen de Docker a un archivo físico (.tar)
# Esto crea el archivo 'dihiggs-base.tar' de ~1GB en tu carpeta actual.
echo "Exportando imagen Docker a archivo..."
docker save dihiggs-base:latest -o dihiggs-base.tar

# 3. Verificar que el archivo existe (ahora sí lo verás)
ls -lh dihiggs-base.tar

# 4. Ahora sí, construir con Apptainer (ya no fallará por 'no such file')
apptainer build dihiggs.sif apptainer.def
